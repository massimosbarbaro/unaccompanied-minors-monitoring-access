Private Sub Comando1_Click()

Dim TemPass As String
Dim ID As Integer
If IsNull(Me.txtLoginID) Then
    MsgBox ("Inserire Login"), vbInformation, "Richiesto Login"
    Me.txtLoginID.SetFocus
ElseIf IsNull(Me.txtPassword) Then
    MsgBox ("Inserire Password"), vbInformation, "Richiesta Password"
    Me.txtPassword.SetFocus
Else
    
        If (IsNull(DLookup("Login", "TabLogin", "Login='" & Me.txtLoginID.Value & "'"))) Or _
        (IsNull(DLookup("Password", "TabLogin", "Password='" & Me.txtPassword.Value & "'"))) Then
            MsgBox ("Login o Password Errati")
        Else
            UserLevel = DLookup("Sicurezza", "TabLogin", "Login = '" & Me.txtLoginID.Value & "'")
            TempPass = DLookup("Password", "TabLogin", "Login = '" & Me.txtLoginID.Value & "'")
            ID = DLookup("LoginID", "TabLogin", "Login = '" & Me.txtLoginID.Value & "'")
            DoCmd.Close
    
            If (TempPass = "Password") Then
              MsgBox ("Per favore cambiare Password"), vbInformation, "Nuova Password"
              DoCmd.OpenForm "CambiaPassword", , , "[LoginID] = " & ID
              Else
                If UserLevel = 1 Then
                    'MsgBox ("Login e Password corretti")
                        DoCmd.OpenForm "fOpen"
                Else
                    If UserLevel = 3 Then
                        'MsgBox ("Login e Password corretti")
                            DoCmd.OpenForm "fOpenAssistenti"
                    Else
                        
                        DoCmd.OpenForm "fPrint"
                    End If
                End If
            End If
        End If
End If

End Sub
