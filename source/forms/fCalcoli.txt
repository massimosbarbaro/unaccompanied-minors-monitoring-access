Private Sub cmdDimissioni_Click()
Dim sqlc As String, rsNum As Recordset, rsRO As Recordset, rsR1 As Recordset
Dim sIdL As String, sqlc1 As String


Set db = CurrentDb
'sqlc = "Select max(IdL) from LPTC"
sqlc = "SELECT Max(CLng(Right([IdL],Len([IdL])-1))) AS L " & _
        "FROM [LPTC];"



Set rsNum = db.OpenRecordset(sqlc, dbOpenSnapshot)
If Not IsNull(rsNum(0)) Then


    sIdL = rsNum(0) + 1
Else
    sIdL = 1
End If


sqlc = "SELECT Go.IdP, Go.Cognome, Go.Nome, Go.DataN, DateAdd('yyyy',18,[datan]) AS Magg " & _
        "FROM (LPTC INNER JOIN [Go] ON LPTC.IdP = Go.IdP) LEFT JOIN 71Dimissione ON LPTC.IdL = [71Dimissione].IdL " & _
        "GROUP BY Go.IdP, Go.Cognome, Go.Nome, Go.DataN, DateAdd('yyyy',18,[datan]), [71Dimissione].IdL " & _
        "HAVING (((Go.DataN)<#12/31/2002#) AND (([71Dimissione].IdL) Is Null));"



sqlc = "SELECT Go.IdP, Go.Cognome, Go.Nome, Go.DataN, DateAdd('yyyy',18,[datan]) AS Magg " & _
        "FROM (LPTC INNER JOIN [Go] ON LPTC.IdP = Go.IdP) LEFT JOIN 71Dimissione ON LPTC.IdP = [71Dimissione].IdP " & _
        "GROUP BY Go.IdP, Go.Cognome, Go.Nome, Go.DataN, DateAdd('yyyy',18,[datan]), [71Dimissione].Cognome " & _
        "HAVING (((Go.DataN)<#12/31/2002#) AND (([71Dimissione].Cognome) Is Null));"




Set rsRO = db.OpenRecordset(sqlc, dbOpenSnapshot)


 sqlc1 = "SELECT * " & _
            "FROM lptc;"
 
 
 'apro il recordset
  Set rsR1 = db.OpenRecordset(sqlc1, dbOpenDynaset)
  
    
'pulisco le variabili
  
    
    Do While Not rsRO.EOF
    
            

             'Carico Tabella Ordinaria
                rsR1.AddNew
                    rsR1!IdL = "C" & sIdL
                    rsR1!IdP = rsRO!IdP
                    rsR1!IdT = 7
                    rsR1!IdC = 0
                    rsR1!Data = rsRO!Magg
                    rsR1!Ora = "00:00"
                    rsR1!Reg = -1
                    rsR1!Note = "Inserimento di ufficio"
                rsR1.Update
                
            sIdL = sIdL + 1
            
    rsRO.MoveNext
    Loop
    
    


rsRO.Close
rsR1.Close
'   cn.Close

Set rsRO = Nothing
Set rsR1 = Nothing
'    Set cn = Nothing










End Sub
