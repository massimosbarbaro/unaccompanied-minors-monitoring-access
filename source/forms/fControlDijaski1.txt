Private Sub cmdOpen_Click()
On Error GoTo Err_cmdOpen_Click

    Dim stDocName As String
    Dim stLinkCriteria As String
    
    Dim sqlc As String
    Dim rsT As Recordset, db As Database

        If IsNull(sIdPr) Or sIdPr = "" Then
            
            MsgBox "Ime??"
            Exit Sub
        
        End If
        

        Set db = CurrentDb
        sqlc = "SELECT IdD, Pass  " & _
                "FROM Dijaski " & _
                "WHERE IdD=" & sIdPr
        
    Set rsT = db.OpenRecordset(sqlc, dbOpenDynaset)

    If Not rsT.EOF Then
        If Not txtGeslo <> rsT("pass") Then
              stDocName = "fOpen"
                DoCmd.OpenForm stDocName, , , stLinkCriteria
        Else
                MsgBox "Geslo?"
        End If
        
    End If

Exit_cmdOpen_Click:
    Exit Sub

Err_cmdOpen_Click:
    MsgBox Err.Description
    Resume Exit_cmdOpen_Click
    
End Sub
