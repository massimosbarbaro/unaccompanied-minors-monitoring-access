Private Sub Form_BeforeInsert(Cancel As Integer)

Dim sqlc As String, rsNum As Recordset
Set db = CurrentDb
'sqlc = "Select max(IdL) from LPTC"
sqlc = "SELECT Max(CLng(Right([IdTp],Len([IdTp])-1))) AS T " & _
        "FROM [Tampone];"



Set rsNum = db.OpenRecordset(sqlc, dbOpenSnapshot)
If Not IsNull(rsNum(0)) Then


    IdTp = "C" & rsNum(0) + 1
Else
    IdTp = 1
End If


If sOpen <> "" Then
    IdP = sOpen
    
Else
    'txtIdConservazione.Enabled = True
End If

End Sub
