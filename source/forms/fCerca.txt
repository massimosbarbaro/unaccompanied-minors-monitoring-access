Private Sub cmdCerca_Click()

sqlc = "SELECT Go.IdP, Go.Sede, Go.Nome, Go.Cognome, Go.Sesso, Go.Cittadinanza, Go.DataN, " & _
            "TipoData.IdT, TipoData.Tipo, LPTC.Data, LPTC.Ora, Comunita.Comunita, LPTC.Note, " & _
            "Go.IdMinoriMIS, Alias.DataModifica, Alias.Cognome, Alias.Nome, Alias.DataN, " & _
            "Alias.Cittadinanza, Alias.Note, Alias.Attiva " & _
        "FROM ((([Go] LEFT JOIN LPTC ON Go.IdP = LPTC.IdP) LEFT JOIN TipoData ON LPTC.IdT = TipoData.IdT) " & _
            "LEFT JOIN Comunita ON LPTC.IdC = Comunita.IdC) LEFT JOIN Alias ON Go.IdP = Alias.IdP " & _
        "WHERE (((Go.Cognome) Like '" & txtCerca & "') " & _
            "And ((TipoData.IdT) = 1 Or (TipoData.IdT) Is Null)) Or (((TipoData.IdT) = 1 " & _
            "Or (TipoData.IdT) Is Null) And ((Alias.Cognome) Like '" & txtCerca & "')) " & _
            "ORDER BY TipoData.IdT;"
            
'Sull 'evento clik del pulsante 1

 DoCmd.OpenForm "fElencoCerca", , , , acFormReadOnly
DoCmd.Maximize
Forms!fElencoCerca.RecordSource = sqlc

'Forms!"fElencoCerca".RecordSource = sqlc
'DoEvents [felencocerca].RowSource = sqlc
           



End Sub
