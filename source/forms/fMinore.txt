Private Sub cmdFolder_Click()
On Error GoTo errorHandler

 Dim Risposta As Integer
 Dim Titolo, Testo As String
 Dim strPath As String
 Dim dbp     As Access.CurrentProject
 Dim bool    As Boolean
 Dim sData As String, sAnno As String
 Dim sAlias As String
 Dim rsR As Recordset
 
 
Set db = CurrentDb
 
 
 With Application
    Set dbp = .CurrentProject
 End With
sData = Format(Me.DataN, "dd_mm_yyyy")

sAnno = Format((DLookup("Data", "LPTC", "IdP='" & Me.IdP & "' and IdT=1")), "yyyy") & "_"

'sAnno = Format(Date, "yyyy") & "_"

'Controllo Alias
sqlc = "SELECT Alias.IdA, Alias.DataModifica, Alias.IdP, Alias.Cognome, Alias.Nome, Alias.LuogoN, Alias.DataN " & _
        "FROM Alias " & _
        "WHERE (((Alias.DataModifica) Is Null) AND ((Alias.IdP)='" & Me.IdP & "'));"

'Apro il recordset
    
    Set rsR = db.OpenRecordset(sqlc, dbOpenDynaset)
  
    Do While Not rsR.EOF
    'Controllo che non esista già il record
    sAlias = rsR!Cognome & "_" & rsR!Nome & "_" & Format(rsR!DataN, "dd_mm_yyyy")
    rsR.MoveNext
    Loop
    
rsR.Close
'rsR1.Close
'   cn.Close

Set rsR = Nothing
'Set rsR1 = Nothing
If sAlias = "" Then
    strPath = dbp.Path & "\Cartelle_MSNA\" & sAnno & Me.Cognome & "_" & Me.Nome & "_" & sData & "\"

Else
    strPath = dbp.Path & "\Cartelle_MSNA\" & sAnno & sAlias & "\"

End If



    
    If Dir(strPath, vbDirectory) <> "" Then
'        MsgBox "Apertura cartella"
        bool = ((GetAttr(strPath) And vbDirectory) = vbDirectory)
'        Shell "EXPLORER.EXE" & " " & Chr(34) & strPath & Chr(34), vbNormalFocus 'Shell "explorer.exe " & strPath, vbNormalFocus
        Shell "cmd /C start """" /max """ & strPath & """", vbHide
    Else
        Titolo = "Creazione cartella Minore"
        Testo = "State per creare la cartella per il minore: " & Chr$(13) & sAnno & Me.Cognome & "_" & Me.Nome & "_" & sData & Chr$(13) & Chr$(13) & "Siete sicuri?"
        Risposta = MsgBox(Testo, vbYesNo, Titolo)
        If Risposta = 6 Then
            MkDir strPath
            MsgBox "Cartella creata con successo!"
        Else
            MsgBox "Creazione annullata!"
        End If
    End If

ext_errorHandler:
    Set dbp = Nothing
    Exit Sub
   
errorHandler:
    With Err
     Select Case .Number
            Case 53
                    VBA.MsgBox "la cartella non esiste!", vbCritical, "Attenzione"
            Case Else
                MsgBox "ERR#" & .Number _
                    & vbNewLine & .Description _
                    , vbOKOnly Or vbCritical
      End Select
    End With
    Resume ext_errorHandler

End Sub

Sub ControlloData()
    Dim dMagg As String
    dMagg = DateDiff("yyyy", [DataN], Now()) + Int(Format(Now(), "mmdd") < Format([DataN], "mmdd"))
    
    
    Select Case dMagg
    Case 17
        txtAnniM.BackColor = RGB(255, 255, 0)
        txtDataM.BackColor = RGB(255, 255, 0)
        
    Case 18
        txtAnniM.BackColor = RGB(255, 0, 0)
        txtDataM.BackColor = RGB(255, 0, 0)
    Case Else
        txtAnniM.BackColor = RGB(255, 255, 255)
        txtDataM.BackColor = RGB(255, 255, 255)
    
    End Select
    


End Sub
